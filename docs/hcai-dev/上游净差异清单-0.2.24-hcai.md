```bash
git -c core.quotePath=false diff --name-status 5c2982a81b46e2142d0143d2f83c1ce9c4da1d24 --
git -c core.quotePath=false diff --name-status 953645ea6c099ab059e00d8acb8a08fbb588c424 --
```
